from fastapi import FastAPI, WebSocket
import aiogram

app = FastAPI()

@app.websocket("/ws/board/{board_id}")
async def board_websocket(websocket: WebSocket, board_id: str):
    await websocket.accept()
    # Connect client to async broadcast channel
    async for message in websocket.iter_text():
        # Save to SQLite WAL database and relay to Telegram Bot API
        await dispatch_to_telegram(board_id, message)
        await broadcast_to_web(board_id, message)
